    /**
     * Điều khiển 2 Banner tự trượt chuyên nghiệp + hỗ trợ vuốt trên điện thoại
     */
    let activeBannerIndex = 0;
    let bannerTimer = null;

    function updateBannerSlides() {
      const slides = document.querySelectorAll('#heroSlider .banner-slide');
      slides.forEach((slide, idx) => {
        if (idx === activeBannerIndex) {
          slide.classList.add('is-active');
        } else {
          slide.classList.remove('is-active');
        }
      });

      for (let i = 0; i < slides.length; i++) {
        const dot = document.getElementById('bannerDot' + i);
        if (!dot) continue;
        if (i === activeBannerIndex) {
          dot.classList.remove('w-2.5', 'bg-white/45');
          dot.classList.add('w-7', 'bg-brand-gold');
        } else {
          dot.classList.remove('w-7', 'bg-brand-gold');
          dot.classList.add('w-2.5', 'bg-white/45');
        }
      }
    }

    function nextBanner() {
      activeBannerIndex = (activeBannerIndex + 1) % 2;
      updateBannerSlides();
      resetBannerAutoSlide();
    }

    function prevBanner() {
      activeBannerIndex = (activeBannerIndex - 1 + 2) % 2;
      updateBannerSlides();
      resetBannerAutoSlide();
    }

    function goToBanner(idx) {
      activeBannerIndex = idx;
      updateBannerSlides();
      resetBannerAutoSlide();
    }

    function resetBannerAutoSlide() {
      if (bannerTimer) clearInterval(bannerTimer);
      bannerTimer = setInterval(function() {
        activeBannerIndex = (activeBannerIndex + 1) % 2;
        updateBannerSlides();
      }, 5500);
    }

    /**
     * Hỗ trợ thao tác vuốt (swipe) trên màn hình cảm ứng cho 2 Banner
     */
    function initBannerTouchSupport() {
      const bannerEl = document.getElementById('bannerContainer');
      if (!bannerEl) return;

      let touchStartX = 0;
      bannerEl.addEventListener('touchstart', function(e) {
        touchStartX = e.changedTouches[0].screenX;
      }, { passive: true });

      bannerEl.addEventListener('touchend', function(e) {
        const touchEndX = e.changedTouches[0].screenX;
        const diff = touchStartX - touchEndX;
        if (Math.abs(diff) > 45) {
          if (diff > 0) nextBanner();
          else prevBanner();
        }
      }, { passive: true });

      bannerEl.addEventListener('mouseenter', function() {
        if (bannerTimer) clearInterval(bannerTimer);
      });

      bannerEl.addEventListener('mouseleave', function() {
        resetBannerAutoSlide();
      });
    }
